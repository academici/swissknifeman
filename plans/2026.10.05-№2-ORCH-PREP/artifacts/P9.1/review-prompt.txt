"""exec/v1 schema registry, semantic validation and document digests."""
from __future__ import annotations

from hashlib import sha256
import json
from pathlib import Path
import re
from typing import Callable, NamedTuple

from orch import CONTRACT_VERSION
from orch import schema

URN_PREFIX = "urn:swissknifeman:orch:exec:"
KINDS = (
    "execution-input", "run-policy", "role-manifest", "capability-record", "run-event",
    "stage-result", "effect-receipt", "check-evidence", "launch-plan",
)
# Plan bridges add their own stricter phrases.
LIFECYCLE_PHRASES = ("report_api", "report API")


class Issue(NamedTuple):
    path: str
    code: str
    message: str
    level: str = "error"


SEMANTIC: dict[str, Callable[[object], list[Issue]]] = {}


def load_registry() -> schema.Registry:
    """Load the cached registry from this package, independent of the current cwd."""
    directory = Path(__file__).resolve().parents[2] / "schemas" / "exec" / "v1"
    return schema.load_registry(directory, urn_prefix=URN_PREFIX, kinds=KINDS)


def semantic_checks(doc: object, kind: str) -> list[Issue]:
    if kind not in KINDS:
        raise schema.SchemaProfileError(f"unknown schema kind: {kind!r}")
    issues = []
    if isinstance(doc, dict) and doc.get("contract_version") != CONTRACT_VERSION:
        issues.append(Issue("/contract_version", "CONTRACT_VERSION",
                            f"contract_version must equal {CONTRACT_VERSION!r}"))
    check = SEMANTIC.get(kind)
    if check is not None:
        issues.extend(check(doc))
    return issues


def validate(doc: object, kind: str, registry: schema.Registry | None = None) -> list[Issue]:
    """Reject incompatible versions, then collect schema and semantic issues."""
    semantic = semantic_checks(doc, kind)
    # A different contract version cannot be interpreted using exec/v1 schemas.
    if any(issue.code == "CONTRACT_VERSION" for issue in semantic):
        return semantic
    reg = registry if registry is not None else load_registry()
    return [Issue(*issue) for issue in schema.validate(doc, kind, reg)] + semantic


def canonical_digest(doc: dict) -> str:
    """Hash canonical JSON without digest or the derived suffix of input_id."""
    payload = {key: value for key, value in doc.items() if key != "digest"}
    if isinstance(payload.get("input_id"), str):
        payload["input_id"] = payload["input_id"].rsplit("@", 1)[0]
    encoded = json.dumps(payload, sort_keys=True, ensure_ascii=False, separators=(",", ":"))
    return sha256(encoded.encode("utf-8")).hexdigest()


def _walk(value: object, path: str = ""):
    """Yield values with escaped JSON pointers, including nested JSON content."""
    yield path, value
    if isinstance(value, dict):
        for key, child in value.items():
            escaped = key.replace("~", "~0").replace("/", "~1")
            yield from _walk(child, f"{path}/{escaped}")
    elif isinstance(value, list):
        for index, child in enumerate(value):
            yield from _walk(child, f"{path}/{index}")


def _execution_input(doc: object) -> list[Issue]:
    if not isinstance(doc, dict):
        return []
    issues = []
    digest, input_id = doc.get("digest"), doc.get("input_id")
    if isinstance(digest, str) and digest != canonical_digest(doc):
        issues.append(Issue("/digest", "DIGEST_MISMATCH", "digest does not match canonical JSON"))
    if isinstance(input_id, str) and isinstance(digest, str) and not input_id.endswith("@" + digest[:12]):
        issues.append(Issue("/input_id", "INPUT_ID", "input_id must end with @digest[:12]"))
    scope, result_contract = doc.get("scope"), doc.get("result_contract")
    if isinstance(scope, dict) and scope.get("read_only") is True \
            and isinstance(result_contract, dict) and result_contract.get("write_authority") != "none":
        issues.append(Issue("/result_contract/write_authority", "INPUT_AUTHORITY",
                            "read-only input requires write_authority none"))
    for path, value in _walk(doc):
        if not isinstance(value, str) or not any(phrase in value for phrase in LIFECYCLE_PHRASES):
            continue
        if path == "/provenance/generated_by/tool":
            continue
        level = "error"
        parts = path.split("/")
        if len(parts) == 4 and parts[1] == "constraints" and parts[2].isdigit() and parts[3] == "text":
            constraints = doc.get("constraints")
            if isinstance(constraints, list):
                constraint = constraints[int(parts[2])]
                if isinstance(constraint, dict) and isinstance(constraint.get("source"), str) \
                        and constraint["source"].startswith("plan.rules"):
                    level = "warning"
        issues.append(Issue(path, "LIFECYCLE_LEAK", "lifecycle instruction is present in execution input", level))
    acceptance = doc.get("acceptance")
    checks = acceptance.get("checks") if isinstance(acceptance, dict) else None
    if isinstance(checks, list):
        for index, check in enumerate(checks):
            if isinstance(check, dict) and ("cmd" in check) == ("procedure" in check):
                issues.append(Issue(f"/acceptance/checks/{index}", "CHECK_CARRIER",
                                    "check must contain exactly one of cmd or procedure"))
    return issues


def _run_policy(doc: object) -> list[Issue]:
    if not isinstance(doc, dict):
        return []
    issues = []
    for path, value in _walk(doc):
        if isinstance(value, int) and not isinstance(value, bool) and value < 0:
            issues.append(Issue(path, "POLICY_NEGATIVE", "policy integers must be non-negative"))
    if "commit" in doc and doc.get("git_action") == "none":
        issues.append(Issue("/commit", "COMMIT_POLICY_CONFLICT",
                            "commit policy requires git_action commit"))
    stages = doc.get("stages")
    if not isinstance(stages, list):
        return issues
    primary = sum(isinstance(stage, dict) and stage.get("kind") == "primary" for stage in stages)
    if primary != 1:
        issues.append(Issue("/stages", "POLICY_PRIMARY", "policy must have exactly one primary stage"))
    graph = {}
    previous = None
    for index, stage in enumerate(stages):
        if not isinstance(stage, dict):
            continue
        if stage.get("required") is True and stage.get("on_unavailable") != "refuse":
            issues.append(Issue(f"/stages/{index}/on_unavailable", "POLICY_REQUIRED_REFUSE",
                                "required stages must refuse when unavailable"))
        stage_id = stage.get("id")
        if not isinstance(stage_id, str):
            continue
        if stage_id in graph:
            issues.append(Issue(f"/stages/{index}/id", "POLICY_DEPENDS", "stage id is duplicated"))
        depends = stage.get("depends_on", [previous] if previous is not None else [])
        graph[stage_id] = [d for d in depends if isinstance(d, str)] if isinstance(depends, list) else []
        previous = stage_id
    unknown = any(dep not in graph for deps in graph.values() for dep in deps)
    if unknown:
        issues.append(Issue("/stages", "POLICY_DEPENDS", "dependency refers to an unknown stage"))
    # Remove satisfied nodes iteratively; this also avoids recursion limits for long pipelines.
    remaining = {name: set(deps) & graph.keys() for name, deps in graph.items()}
    while remaining:
        ready = {name for name, deps in remaining.items() if not deps}
        if not ready:
            issues.append(Issue("/stages", "POLICY_DEPENDS", "stage dependencies contain a cycle"))
            break
        remaining = {name: deps - ready for name, deps in remaining.items() if name not in ready}
    review = doc.get("review")
    if isinstance(review, dict):
        for field, limit in (("max_rounds", 2), ("max_corrections", 1)):
            value = review.get(field)
            if isinstance(value, int) and not isinstance(value, bool) and value > limit:
                issues.append(Issue(f"/review/{field}", "POLICY_REVIEW_LIMIT", f"{field} must not exceed {limit}"))
        has_review = any(isinstance(s, dict) and s.get("kind") == "review" for s in stages)
        if isinstance(review.get("enabled"), bool) and review["enabled"] != has_review:
            issues.append(Issue("/review/enabled", "POLICY_REVIEW_STAGE",
                                "review.enabled must match the presence of a review stage"))
    return issues


def _role_manifest(doc: object) -> list[Issue]:
    if not isinstance(doc, dict) or not isinstance(doc.get("tools"), dict):
        return []
    authority, tools = doc.get("write_authority"), doc["tools"]
    if (authority == "plan") != (tools.get("plan_api") == "write") \
            or (authority == "none" and tools.get("filesystem") == "write"):
        return [Issue("/write_authority", "ROLE_AUTHORITY", "write authority conflicts with tool permissions")]
    return []


def _capability_record(doc: object) -> list[Issue]:
    if not isinstance(doc, dict) or doc.get("status") != "declared":
        return []
    evidence = doc.get("evidence")
    if isinstance(evidence, list) and any(isinstance(e, dict) and e.get("kind") in ("doc", "code")
                                          for e in evidence):
        return []
    fields = ("model", "effort", "stream", "attach", "cancel", "resume", "usage",
              "restrictions", "skill_delivery", "plan_write_prevention")
    for field in fields:
        value = doc.get(field)
        values = value.items() if isinstance(value, dict) else (("", value),)
        for name, support in values:
            if name in ("notes", "mode", "prompt_only"):
                continue
            if support == "supported":
                return [Issue("/evidence", "CAPABILITY_EVIDENCE",
                              "declared supported capabilities require doc or code evidence")]
    return []


SEMANTIC.update({
    "execution-input": _execution_input,
    "run-policy": _run_policy,
    "role-manifest": _role_manifest,
    "capability-record": _capability_record,

def _attempt_range(doc: dict, *, event: bool = False) -> list[Issue]:
    issues = []
    for name, minimum in (("attempt", 1), ("seq", 0)) if event else (("attempt", 1),):
        value = doc.get(name)
        if isinstance(value, int) and not isinstance(value, bool) and value < minimum:
            issues.append(Issue(f"/{name}", "ATTEMPT_RANGE", f"{name} must be at least {minimum}"))
    return issues


def _run_event(doc: object) -> list[Issue]:
    return _attempt_range(doc, event=True) if isinstance(doc, dict) else []


def _stage_result(doc: object) -> list[Issue]:
    if not isinstance(doc, dict):
        return []
    issues = _attempt_range(doc)
    commits = doc.get("commits")
    if isinstance(commits, list):
        seen = set()
        for index, commit in enumerate(commits):
            commit_sha = commit.get("sha") if isinstance(commit, dict) else None
            if not isinstance(commit_sha, str):
                continue
            if commit_sha in seen:
                issues.append(Issue(f"/commits/{index}/sha", "COMMIT_SHA_DUPLICATE",
                                    "commit hashes must be unique"))
            seen.add(commit_sha)
    usage = doc.get("usage")
    if isinstance(usage, dict):
        if usage.get("status") == "unknown" and any(
                name in usage for name in ("input_tokens", "output_tokens", "cost_usd")):
            issues.append(Issue("/usage", "USAGE_UNKNOWN_NUMERIC",
                                "unknown usage must not include token counts or cost"))
        cost = usage.get("cost_usd")
        if isinstance(cost, str) and re.fullmatch(r"\d+(\.\d+)?", cost, re.ASCII) is None:
            issues.append(Issue("/usage/cost_usd", "COST_FORMAT", "cost_usd must be a decimal string"))
    errors = doc.get("errors")
    if doc.get("status") == "completed" and isinstance(errors, list):
        for index, error in enumerate(errors):
            if isinstance(error, dict) and error.get("class") == "parse":
                issues.append(Issue(f"/errors/{index}", "COMPLETED_PARSE_ERROR",
                                    "completed stage cannot contain a parse error"))
    return issues


def _effect_receipt(doc: object) -> list[Issue]:
    if not isinstance(doc, dict):
        return []
    reconcile = doc.get("reconcile")
    if doc.get("outcome") == "unknown" and isinstance(reconcile, dict) and reconcile.get("method") == "none":
        return [Issue("/reconcile/method", "RECEIPT_RECONCILE",
                      "unknown effect outcome requires a reconciliation method")]
    return []


def _check_evidence(doc: object) -> list[Issue]:
    if not isinstance(doc, dict) or not isinstance(doc.get("procedure"), dict):
        return []
    procedure = doc["procedure"]
    command, manual = "cmd" in procedure, "procedure" in procedure
    if command == manual:
        return [Issue("/procedure", "EVIDENCE_CARRIER", "exactly one of cmd or procedure is required")]
    if command and ("exit" not in doc or "verdict" in doc):
        return [Issue("/procedure", "EVIDENCE_CARRIER", "command evidence requires exit without verdict")]
    if manual and ("verdict" not in doc or "observation" not in doc or "exit" in doc):
        return [Issue("/procedure", "EVIDENCE_CARRIER",
                      "manual evidence requires verdict and observation without exit")]
    return []


def _launch_plan(doc: object) -> list[Issue]:
    if not isinstance(doc, dict) or not isinstance(doc.get("stages"), list):
        return []
    issues = []
    for index, stage in enumerate(doc["stages"]):
        if not isinstance(stage, dict) or not isinstance(stage.get("env"), dict):
            continue
        for name, value in stage["env"].items():
            if not isinstance(value, str):
                escaped = name.replace("~", "~0").replace("/", "~1")
                issues.append(Issue(f"/stages/{index}/env/{escaped}", "type",
                                    "environment variable value must be a string"))
    return issues

"""Closed-profile JSON Schema validator (stdlib only).

Adapted copy of the plan store's closed-profile validator (PLAN2.D4);
the plan bridge tests keyword parity.

Supports exactly the keywords in ``SUPPORTED_KEYWORDS``; any other keyword in a schema file fails
closed with ``SchemaProfileError`` at load time. Validation collects every issue instead of stopping at the
first one.
"""
from __future__ import annotations

import json
from pathlib import Path
import re
from typing import Any, NamedTuple

DIALECT = "https://json-schema.org/draft/2020-12/schema"
ERROR_CODES = ("type", "enum", "const", "required", "additional", "pattern", "min_items", "unique",
               "min_length", "one_of", "ref")
SUPPORTED_KEYWORDS = frozenset({
    "$schema", "$id", "$defs", "$ref", "title", "description", "type", "enum", "const", "properties",
    "required", "additionalProperties", "items", "minItems", "uniqueItems", "minLength", "pattern", "oneOf",
    "default",
})
_TYPE_NAMES = frozenset({"null", "boolean", "integer", "number", "string", "array", "object"})
_cache: dict[tuple[Path, str, tuple[str, ...]], "Registry"] = {}


class SchemaProfileError(ValueError):
    """A schema file leaves the closed profile, or an unknown schema kind was requested."""


class RefError(LookupError):
    """A ``$ref`` that the registry cannot resolve."""


class ValidationIssue(NamedTuple):
    path: str  # JSON pointer such as "/reads/2/path"; "" is the document root
    code: str
    message: str


class Registry:
    """Loaded schemas by ``$id``; read-only after construction."""

    def __init__(self, schemas: dict[str, dict], *, urn_prefix: str, kinds: tuple[str, ...]):
        self._schemas = schemas
        self.urn_prefix = urn_prefix
        self.kinds = tuple(kinds)

    def for_kind(self, kind: str) -> tuple[dict, str]:
        schema_id = f"{self.urn_prefix}{kind}:v1"
        if kind not in self.kinds or schema_id not in self._schemas:
            raise SchemaProfileError(f"unknown schema kind: {kind!r}")
        return self._schemas[schema_id], schema_id

    def resolve(self, ref: str, base_id: str) -> tuple[dict, str]:
        """Resolve ``#/$defs/x`` or ``<id>#/$defs/x``; return the node and the id of its owning schema."""
        document, _, fragment = ref.partition("#")
        owner = document or base_id
        node: Any = self._schemas.get(owner)
        if node is None:
            raise RefError(f"unknown schema {owner!r} in $ref {ref!r}")
        if fragment:
            if not fragment.startswith("/$defs/") or fragment.count("/") != 2:
                raise RefError(f"unsupported $ref fragment in {ref!r}")
            node = node.get("$defs", {}).get(fragment[len("/$defs/"):])
            if not isinstance(node, dict):
                raise RefError(f"$ref {ref!r} does not resolve")
        return node, owner


def load_registry(schema_dir: Path, *, urn_prefix: str, kinds: tuple[str, ...]) -> Registry:
    """Load and profile-check schemas; cache by directory and registry parameters."""
    directory = Path(schema_dir).resolve()
    key = (directory, urn_prefix, tuple(kinds))
    cached = _cache.get(key)
    if cached is not None:
        return cached
    schemas: dict[str, dict] = {}
    for path in sorted(directory.glob("*.schema.json")):
        try:
            schema = json.loads(path.read_text(encoding="utf-8"))
        except (OSError, ValueError) as exc:
            raise SchemaProfileError(f"{path.name}: cannot load schema: {exc}") from exc
        _check_profile(schema, path.name, "", root=True)
        schema_id = schema.get("$id")
        if not isinstance(schema_id, str) or not schema_id:
            raise SchemaProfileError(f"{path.name}: missing $id")
        if schema_id in schemas:
            raise SchemaProfileError(f"{path.name}: duplicate $id {schema_id!r}")
        schemas[schema_id] = schema
    if not schemas:
        raise SchemaProfileError(f"no *.schema.json files in {directory}")
    registry = Registry(schemas, urn_prefix=urn_prefix, kinds=tuple(kinds))
    _cache[key] = registry
    return registry


def clear_registry_cache() -> None:
    _cache.clear()


def _check_profile(node: Any, where: str, pointer: str, *, root: bool = False) -> None:
    """Reject keywords outside the closed profile and malformed keyword values."""
    def fail(message: str) -> None:
        raise SchemaProfileError(f"{where}: {message} at {pointer or '/'}")

    if not isinstance(node, dict):
        fail("schema node must be an object")
    for keyword, value in node.items():
        if keyword not in SUPPORTED_KEYWORDS:
            fail(f"unsupported keyword {keyword!r}")
        if keyword == "$schema" and (not root or value != DIALECT):
            fail(f"$schema must be {DIALECT!r} on the root only")
        elif keyword == "additionalProperties" and value is not False:
            fail("additionalProperties may only be false")
        elif keyword == "type":
            names = [value] if isinstance(value, str) else value
            if not isinstance(names, list) or not names or any(n not in _TYPE_NAMES for n in names):
                fail(f"invalid type {value!r}")
        elif keyword == "required" and not (isinstance(value, list) and all(isinstance(n, str) for n in value)):
            fail("required must be a list of strings")
        elif keyword == "enum" and not (isinstance(value, list) and value):
            fail("enum must be a non-empty list")
        elif keyword in ("minItems", "minLength") and not (
                isinstance(value, int) and not isinstance(value, bool) and value >= 0):
            fail(f"{keyword} must be a non-negative integer")
        elif keyword == "uniqueItems" and not isinstance(value, bool):
            fail("uniqueItems must be a boolean")
        elif keyword == "pattern":
            try:
                _compile(value)
            except (re.error, TypeError) as exc:
                fail(f"invalid pattern {value!r}: {exc}")
        elif keyword == "$ref" and not isinstance(value, str):
            fail("$ref must be a string")
        elif keyword in ("properties", "$defs"):
            if not isinstance(value, dict):
                fail(f"{keyword} must be an object")
            for name, child in value.items():
                _check_profile(child, where, f"{pointer}/{keyword}/{name}")
        elif keyword == "items":
            _check_profile(value, where, f"{pointer}/items")
        elif keyword == "oneOf":
            if not isinstance(value, list) or not value:
                fail("oneOf must be a non-empty list")
            for index, child in enumerate(value):
                _check_profile(child, where, f"{pointer}/oneOf/{index}")


def _compile(pattern: str) -> re.Pattern:
    """JSON Schema (ECMA-262) semantics: ASCII classes, and ``$`` only at the very end of the string."""
    if pattern.endswith("$") and not pattern.endswith("\\$"):
        pattern = pattern[:-1] + r"\Z"
    return re.compile(pattern, re.ASCII)


def validate(doc: object, kind: str, registry: Registry) -> list[ValidationIssue]:
    """Validate ``doc`` against the schema of ``kind``; return every issue found."""
    reg = registry
    schema, schema_id = reg.for_kind(kind)
    issues: list[ValidationIssue] = []
    _check(reg, schema, schema_id, doc, "", issues)
    return issues


def validate_node(registry: Registry, node: dict, base_id: str, value: object) -> list[ValidationIssue]:
    """Validate ``value`` against one schema node (for example a ``oneOf`` branch) owned by ``base_id``."""
    issues: list[ValidationIssue] = []
    _check(registry, node, base_id, value, "", issues)
    return issues


def _check(reg: Registry, schema: dict, base: str, value: Any, pointer: str, out: list[ValidationIssue]) -> None:
    def add(code: str, message: str, at: str | None = None) -> None:
        location = pointer if at is None else at
        out.append(ValidationIssue(location, code, f"{location or '/'}: {message}"))

    if "$ref" in schema:
        try:
            target, owner = reg.resolve(schema["$ref"], base)
        except RefError as exc:
            add("ref", str(exc))
        else:
            _check(reg, target, owner, value, pointer, out)
    names = schema.get("type")
    if names is not None:
        names = [names] if isinstance(names, str) else names
        if not any(_is_type(value, n) for n in names):
            add("type", f"expected {' or '.join(names)}, got {_type_name(value)}")
            return
    if "enum" in schema and not any(_equal(value, option) for option in schema["enum"]):
        add("enum", "must be one of " + ", ".join(json.dumps(o, ensure_ascii=False) for o in schema["enum"]))
    if "const" in schema and not _equal(value, schema["const"]):
        add("const", f"must equal {json.dumps(schema['const'], ensure_ascii=False)}")
    if isinstance(value, str):
        if len(value) < schema.get("minLength", 0):
            add("min_length", f"string is shorter than {schema['minLength']}")
        pattern = schema.get("pattern")
        if pattern is not None and _compile(pattern).search(value) is None:
            add("pattern", f"{_short(value)} does not match pattern {pattern}")
    elif isinstance(value, list):
        _check_array(reg, schema, base, value, pointer, out, add)
    elif isinstance(value, dict):
        _check_object(reg, schema, base, value, pointer, out, add)
    branches = schema.get("oneOf")
    if branches is not None:
        matched = 0
        for branch in branches:
            trial: list[ValidationIssue] = []
            _check(reg, branch, base, value, pointer, trial)
            matched += not trial
        if matched != 1:
            add("one_of", f"matches {matched} of {len(branches)} oneOf branches, expected exactly 1")


def _check_array(reg, schema, base, value, pointer, out, add) -> None:
    if len(value) < schema.get("minItems", 0):
        add("min_items", f"array has fewer than {schema['minItems']} items")
    if schema.get("uniqueItems"):
        for j, later in enumerate(value):
            first = next((i for i in range(j) if _equal(value[i], later)), None)
            if first is not None:
                add("unique", f"items {first} and {j} are equal")
                break
    item_schema = schema.get("items")
    if item_schema is not None:
        for index, element in enumerate(value):
            _check(reg, item_schema, base, element, f"{pointer}/{index}", out)


def _check_object(reg, schema, base, value, pointer, out, add) -> None:
    for name in schema.get("required", ()):
        if name not in value:
            add("required", f"missing required property {name!r}")
    properties = schema.get("properties", {})
    for name, child in properties.items():
        if name in value:
            _check(reg, child, base, value[name], f"{pointer}/{_escape(name)}", out)
    if schema.get("additionalProperties") is False:
        for name in value:
            if name not in properties:
                add("additional", f"unexpected property {name!r}", f"{pointer}/{_escape(name)}")


def _escape(key: str) -> str:
    return key.replace("~", "~0").replace("/", "~1")


def _is_type(value: Any, name: str) -> bool:
    if name == "null":
        return value is None
    if name == "boolean":
        return isinstance(value, bool)
    if name == "integer":
        return isinstance(value, int) and not isinstance(value, bool)
    if name == "number":
        return isinstance(value, (int, float)) and not isinstance(value, bool)
    if name == "string":
        return isinstance(value, str)
    if name == "array":
        return isinstance(value, list)
    return isinstance(value, dict)  # "object"


def _type_name(value: Any) -> str:
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "boolean"
    return {int: "integer", float: "number", str: "string", list: "array", dict: "object"}.get(
        type(value), type(value).__name__)


def _equal(a: Any, b: Any) -> bool:
    """JSON equality: booleans never equal numbers; ints and floats compare by value."""
    if isinstance(a, bool) or isinstance(b, bool):
        return isinstance(a, bool) and isinstance(b, bool) and a == b
    if isinstance(a, list) and isinstance(b, list):
        return len(a) == len(b) and all(_equal(x, y) for x, y in zip(a, b))
    if isinstance(a, dict) and isinstance(b, dict):
        return a.keys() == b.keys() and all(_equal(a[k], b[k]) for k in a)
    if isinstance(a, (int, float)) and isinstance(b, (int, float)):
        return a == b
    return type(a) is type(b) and a == b


def _short(value: str) -> str:
    text = json.dumps(value, ensure_ascii=False)
    return text if len(text) <= 60 else text[:57] + '..."'

from hashlib import sha256

class SchemaProfileTests(unittest.TestCase):
    def test_unsupported_keyword_fails_closed(self):
        with tempfile.TemporaryDirectory() as directory:
            Path(directory, "bad.schema.json").write_text(json.dumps({
                "$id": "urn:test:bad:v1", "type": "integer", "minimum": 0,
            }))
            with self.assertRaisesRegex(schema.SchemaProfileError, "unsupported keyword 'minimum'"):
                schema.load_registry(Path(directory), urn_prefix="urn:test:", kinds=("bad",))

    def test_digest_canonical_bytes_and_no_mutation(self):
        original = {"digest": "old", "z": "привет", "a": {"digest": "nested", "n": 1}}
        reordered = {"a": {"n": 1, "digest": "nested"}, "z": "привет", "digest": "new"}
        expected = sha256('{"a":{"digest":"nested","n":1},"z":"привет"}'.encode()).hexdigest()
        self.assertEqual(contracts.canonical_digest(original), expected)
        self.assertEqual(contracts.canonical_digest(reordered), expected)
        self.assertEqual(original["digest"], "old")
        self.assertNotEqual(contracts.canonical_digest({"a": {"n": 1}, "z": "привет"}), expected)

    def test_shared_definitions(self):
        registry = contracts.load_registry()
        base = contracts.URN_PREFIX + "defs:v1"
        valid = {
            "id": "PLAN2.P1.1/U1@run", "sha256": "a" * 64, "gitSha": "abc1234",
            "ts": "2026-10-05T17:00:00.123Z", "text": "x", "path": "@plan/knowledge/a.json",
            "provider": "codex", "modelClass": "implementation", "effort": "high",
            "route": {"class": "implementation", "effort": "high"},
            "actor": {"provider": "codex", "model": "gpt-6.1-sol"},
            "check": {"id": "V1", "cmd": "true", "expect": "exit 0", "kind": "acceptance"},
            "support": "unknown", "binding": "cao", "roleId": "reviewer", "commitRole": "fix",
            "section": {"id": "x", "title": "X", "text": ""},
        }
        root, _ = registry.resolve(base, base)
        self.assertEqual(set(root["$defs"]), set(valid))
        for name, value in valid.items():
            with self.subTest(name=name):
                node = {"$ref": "#/$defs/" + name}
                self.assertEqual(schema.validate_node(registry, node, base, value), [])
                self.assertTrue(schema.validate_node(registry, node, base, None))
        for name, value in (("id", "x" * 161), ("id", "abc\n"), ("sha256", "A" * 64),
                            ("path", "/absolute"), ("path", "has space"), ("provider", "Codex"),
                            ("route", {"class": "economy"}),
                            ("actor", {"provider": "codex", "model": "x", "extra": 1})):
            with self.subTest(name=name, value=value):
                self.assertTrue(schema.validate_node(registry, {"$ref": "#/$defs/" + name}, base, value))
        for definition in root["$defs"].values():
            for prop in definition.get("properties", {}).values():
                self.assertTrue(prop.get("description"))

    def test_registry_cache_includes_route_parameters(self):

    def test_schema_and_semantic_dispatch(self):
        kind = "execution-input"
        registry = schema.Registry({contracts.URN_PREFIX + kind + ":v1": {
            "type": "object", "required": ["goal"],
        }}, urn_prefix=contracts.URN_PREFIX, kinds=contracts.KINDS)
        semantic = contracts.Issue("/goal", "TEST_SEMANTIC", "test issue")
        with patch.dict(contracts.SEMANTIC, {kind: lambda doc: [semantic]}):
            issues = contracts.validate({"contract_version": "exec/v1"}, kind, registry)
        self.assertEqual([issue.code for issue in issues], ["required", "TEST_SEMANTIC"])
        self.assertTrue(all(issue.level == "error" for issue in issues))
        with self.assertRaises(schema.SchemaProfileError):
            contracts.validate({"contract_version": "exec/v1"}, "unknown", registry)
